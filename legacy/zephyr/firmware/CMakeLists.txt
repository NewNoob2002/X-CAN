cmake_minimum_required(VERSION 3.20.0)
list(APPEND BOARD_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
find_package(Zephyr 4.4.0 REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(xcan)
set(XCAN_SHARED "${CMAKE_CURRENT_SOURCE_DIR}/../../../firmware/src")
target_include_directories(app PRIVATE ${XCAN_SHARED})
target_sources(app PRIVATE src/main.c)
if(CONFIG_XCAN_CAN)
  target_sources(app PRIVATE src/usb_can.c src/can_app.c src/can_port.c ${XCAN_SHARED}/protocol_v2.c)
  include(mcan.cmake)
else()
  target_sources_ifdef(CONFIG_XCAN_USB app PRIVATE src/usb.c ${XCAN_SHARED}/protocol.c)
endif()
